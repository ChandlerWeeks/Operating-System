18#[panic_handler]
19fn panic(info: &core::panic::PanicInfo<'_>) -> ! {
20    print!("[\x1b[91mOS PANIC\x1b[0m]");
21    if let Some(loc) = info.location() {
22        print!(" {}@{}.{}", loc.file(), loc.line(), loc.column());
23    }
24    println!(": {}", info.message());
25    // Busy loop to prevent panic!() from returning.
26    loop {
27        unsafe {
28            core::arch::asm!("wfi");
29        }
30    }
31}

37/// # Entry point
38///
39/// Coming from the Limine boot loader, we are in S-mode with the MMU turned on.
40#[unsafe(no_mangle)]
41unsafe extern "C" fn _start() -> ! {
42    unsafe extern "C" {
43        #[link_name = "__global_pointer$"]
44        unsafe static __GLOBAL_POINTER: u8;
45    }
46
47    unsafe {
48        // Set the global pointer.
49        core::arch::asm!(
50            "la gp, {address}",
51            address = sym __GLOBAL_POINTER,
52            options(nostack, nomem),
53        );
54    }
55    main();
56    sbi::hart_stop();
57}

59fn main() {
60    // Print the ACPI hardware values before the system shuts down.
61    clear_screen();
62
63    let Some(rsdp_vaddr) = limine::rsdp_virt() else {
64        debugln!("ACPI: Limine did not provide an RSDP address.");
65        sbi::shutdown();
66    };
67    let Some(xsdt) = acpi::Xsdt::from_rsdp(rsdp_vaddr as usize) else {
68        debugln!("ACPI: The RSDP does not contain a valid XSDT address.");
69        sbi::shutdown();
70    };
71
72    let mut aplic_address = None;
73    let mut imsic_address = None;
74    let mut ecam_address = None;
75    let mut uart_address = None;
76    let mut timer_frequency = None;
77    let mut imsic_printed = false;
78
79    for table in xsdt.iter() {
80        if let Some(madt) = table.madt() {
81            let mut index = 0;
82            while let Some(entry) = madt.get_entry(index) {
83                match entry {
84                    MadtStructure::Rintc(rintc) => {
85                        let address = rintc.imsic_base_address();
86                        imsic_address.get_or_insert(address);
87                        debugln!(
88                            "HART #{} IMSIC @ {:#018X}",
89                            rintc.hart_id(),
90                            limine::pa_to_va(address as usize)
91                        );
92                    }
93                    MadtStructure::Imsic(imsic) if !imsic_printed => {
94                        debugln!(
95                            "IMSIC: version {}, supported IDs {}, guest IDs {}, hart index bits {}",
96                            imsic.version(),
97                            imsic.num_sup_interrupt_ids(),
98                            imsic.num_guest_interrupt_ids(),
99                            imsic.hart_index_bits()
100                        );
101                        imsic_printed = true;
102                    }
103                    MadtStructure::Aplic(aplic) if aplic_address.is_none() => {
104                        let address = aplic.aplic_address();
105                        aplic_address = Some(address);
106                        debugln!(
107                            "APLIC: VA {:#018X}, PA {:#010X}",
108                            limine::pa_to_va(address as usize),
109                            address
110                        );
111                    }
112                    _ => {}
113                }
114                index += 1;
115            }
116        }
117
118        if let Some(mcfg) = table.mcfg() {
119            ecam_address = mcfg.get_entry(0).map(|entry| entry.base);
120        }
121
122        if let Some(spcr) = table.spcr() {
123            if spcr.address_space_id() == 0 {
124                let address = spcr.base_address();
125                uart_address = Some(address);
126                debugln!(
127                    "UART: {:#018X}, freq: {}, irq: {}, irq_type: {}",
128                    limine::pa_to_va(address as usize),
129                    spcr.uart_frequency(),
130                    spcr.irq(),
131                    spcr.interrupt_type()
132                );
133            } else {
134                debugln!("ACPI SPCR: UART address is not in system memory.");
135            }
136        }
137
138        if let Some(rhct) = table.rhct() {
139            timer_frequency = Some(rhct.time_base_freq());
140            debugln!("RHCT: {:?}", rhct);
141        }
142    }
143
144    print_address("PCI ECAM", ecam_address);
145    print_frequency(timer_frequency);
146
147    if aplic_address.is_none() {
148        debugln!("ACPI APLIC address: not found");
149    }
150    if imsic_address.is_none() {
151        debugln!("ACPI IMSIC address: not found");
152    }
153    if uart_address.is_none() {
154        debugln!("ACPI UART address: not found");
155    }
156
157    sbi::shutdown();
158}

160/// Prints one detected physical address.
161fn print_address(name: &str, address: Option<u64>) {
162    match address {
163        Some(address) => debugln!("ACPI {} address: {:#018X}", name, address),
164        None => debugln!("ACPI {} address: not found", name),
165    }
166}
167
168/// Prints the detected timer frequency.
169fn print_frequency(frequency: Option<u64>) {
170    match frequency {
171        Some(frequency) => debugln!("ACPI timer frequency: {} Hz", frequency),
172        None => debugln!("ACPI timer frequency: not found"),
173    }
174}
