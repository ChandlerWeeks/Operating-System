123/// Get the HHDM offset.
124fn hhdm_offset() -> u64 {
125    let ptr = HHDM_REQUEST.response.load(Ordering::Relaxed);
126    assert!(!ptr.is_null(), "Limine did not provide HHDM response");
127    unsafe { (*ptr).offset }
128}
129
130/// Convert an HHDM physical address to a virtual address.
131fn hhdm_phys_to_virt(paddr: usize) -> usize {
132    let off = hhdm_offset() as usize;
133    off.saturating_add(paddr)
134}
135/// Convert an HHDM virtual address to a physical address.
136fn hhdm_virt_to_phys(vaddr: usize) -> usize {
137    let off = hhdm_offset() as usize;
138    vaddr.saturating_sub(off)
139}

180/// Convert a kernel virtual address to a physical address.
181pub fn va_to_pa(addr: usize) -> usize {
182    if addr & 0xFFFF_FFFF_0000_0000 == 0xFFFF_FFFF_0000_0000 {
183        // Kernel executable address
184        let kv = kernel_virt_base();
185        let kp = kernel_phys_base();
186        (kp + (addr as u64 - kv)) as usize
187    } else if addr & config::KERNEL_IO_ADDR == config::KERNEL_IO_ADDR {
188        // This is an IO mapping
189        addr.saturating_sub(config::KERNEL_IO_ADDR)
190    } else if addr & config::KERNEL_HEAP_ADDR == config::KERNEL_HEAP_ADDR {
191        addr.saturating_sub(config::KERNEL_HEAP_ADDR)
192    } else if addr & 0xFFFF_8000_0000_0000 == 0xFFFF_8000_0000_0000 {
193        // HHDM mapped virtual address
194        hhdm_virt_to_phys(addr)
195    } else {
196        // Lower-half, so already a physical address.
197        addr
198    }
199}
200
201/// Convert a physical address to a kernel virtual address.
202pub fn pa_to_va(addr: usize) -> usize {
203    if addr & 0xFFFF_8000_0000_0000 == 0xFFFF_8000_0000_0000 {
204        // Address is already a virtual address, so just return it.
205        return addr;
206    }
207    if addr < 0x8000_0000 {
208        // This is an MMIO address.
209        return config::KERNEL_IO_ADDR + addr;
210    }
211    // HHDM mapped virtual address
212    hhdm_phys_to_virt(addr)
213}

290/// Get the Limine command line as UTF-8 text.
291pub fn cmd_line() -> Option<&'static str> {
292    let ptr = CMD_LINE_REQUEST.response.load(Ordering::SeqCst);
293    if ptr.is_null() {
294        return None;
295    }
296    let response = unsafe { ptr.as_ref_unchecked() };
297    match unsafe { core::ffi::CStr::from_ptr(response.cmdline) }.to_str() {
298        Ok(x) => Some(x),
299        Err(_) => None,
300    }
301}

350/// Iterate over usable physical memory regions.
351pub fn usable_memory_regions() -> impl Iterator<Item = (u64, u64)> {
352    let ptr = MEMORY_MAP_REQUEST.response.load(Ordering::Acquire);
353    assert!(!ptr.is_null(), "Limine did not provide memory map");
354
355    let response = unsafe { &*ptr };
356    let count = response.entry_count as usize;
357    let entries = response.entries;
358
359    (0..count).filter_map(move |i| {
360        let entry = unsafe { &**entries.add(i) };
361        if entry.entry_type == memmap::USABLE {
362            Some((entry.base, entry.length))
363        } else {
364            None
365        }
366    })
367}
368
369/// Iterate over all memory map entries.
370pub fn all_memory_regions() -> impl Iterator<Item = &'static MemoryMapEntry> {
371    let ptr = MEMORY_MAP_REQUEST.response.load(Ordering::Acquire);
372    assert!(!ptr.is_null(), "Limine did not provide memory map");
373
374    let response = unsafe { &*ptr };
375    let count = response.entry_count as usize;
376    let entries = response.entries;
377
378    (0..count).map(move |i| unsafe { &**entries.add(i) })
379}
