8/// # Kernel's Output
9pub struct KernelOut;
10/// implements the write trait from core::fmt for KernelOut, 
11/// allowing the write system call to a place in the system.
12impl core::fmt::Write for KernelOut {
13    /// processes each byte (character) in a string and outputs it using the SBI
14    ///
15    /// Inputs:
16    ///
17    /// - `s`: the string to be output
18    ///
19    /// Return:
20    ///
21    /// - `result`: Successful run of the function
22    fn write_str(&mut self, s: &str) -> core::fmt::Result {
23        for c in s.bytes() {
24            crate::sbi::debug_write_char(c);
25        }
26        Ok(())
27    }
28}

45/// # Print Line Macro

49/// Input Options:
50///
51/// () => outputs a newline character and nothing else
52/// ($fmt:expr) => outputs one string and a newline character

54#[macro_export]
55macro_rules! println {
56    () => (
57        print!("\r\n")

59    ($fmt:expr) => (
60        print!(concat!($fmt, "\r\n"))
61    );
62    ($fmt:expr, $($args:tt)+) => (
63        print!(concat!($fmt, "\r\n"), $($args)+)
64    );

72/// Kernel output with auto indentation for line feed
73pub struct DebugOutput;
74/// implements the write trait from core::fmt for KernelOut, 
75/// allowing the write system call to a place in the system.
76impl core::fmt::Write for DebugOutput {
77    /// processes each byte (character) in a string and outputs it using the SBI
78    ///
79    /// Inputs:
80    ///
81    /// - `s`: the string to be output
82    ///
83    /// Return:
84    ///
85    /// - `result`: Successful run of the function
86    fn write_str(&mut self, s: &str) -> core::fmt::Result {
87        for byte in s.bytes() {
88            crate::sbi::debug_write_char(byte);
89            // auto indent on a new line for debugging lines.
90            if byte == b'\n' {
91                crate::sbi::debug_write_char(b' ');
92                crate::sbi::debug_write_char(b' ');
93            }
94        }
95        Ok(())
96    }
97}

99/// Prints source code information with an optional message
100#[macro_export]
101macro_rules! debugln
102{
103    // No message: print a file name and line number
104    () => ({
105        const LINE: u32 = line!();
106        const FILE: &'static str = $crate::print::file_name(file!());
107        use core::fmt::Write;
108        let mut x = $crate::print::DebugOutput; let _ = write!(x, "[\x1b[96m{:<12} @ {:>4}\x1B[0m]", FILE, LINE);
109        $crate::sbi::debug_write_char(b'\n');
110    });
111    // Takes one or many arguements as a message: Prints a message and a file name with a line number
112    ($($args:tt)+) => ({
113        const LINE: u32 = line!();
114        const FILE: &'static str = $crate::print::file_name(file!());
115        use core::fmt::Write;
116        let mut x = $crate::print::DebugOutput;
117        let _ = write!(x, "[\x1b[96m{:<12} @ {:>4}\x1B[0m]\n", FILE, LINE);
118        let _ = write!(x, $($args)+);
119        $crate::sbi::debug_write_char(b'\n');
120    });
121}

123/// Reads and returns a file name from a directory path by finding the items after the last / 
124pub const fn file_name(path: &str) -> &str {
125    let bytes = path.as_bytes();
126    let mut i = bytes.len() - 1;
127    while i > 0 {
128        if bytes[i] == b'/' {
129            return unsafe {
130                core::str::from_utf8_unchecked(
131                    bytes.split_at(i + 1).1
132                )   
133            }
134        }
135        i -= 1
136    }
137    path
138}
